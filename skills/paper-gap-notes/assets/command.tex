\newcommand{\Lean}{\textsc{Lean}}

% A formal declaration name, upright sans, breakable at underscores.
\ExplSyntaxOn
\NewDocumentCommand{\leanid}{m}{
  \ifmmode
    \textsf{\detokenize{#1}}
  \else
    \group_begin:
    \urlstyle{sf}
    \tl_set:Nn \l_tmpa_tl {#1}
    \tl_replace_all:Nnn \l_tmpa_tl {\_} {_}
    \exp_args:NV \path \l_tmpa_tl
    \group_end:
  \fi
}
\ExplSyntaxOff
